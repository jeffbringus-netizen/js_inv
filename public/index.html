SELECT
    p.product_id,
    p.model,
    p.quantity,
    p.status,
    pd.name
FROM oc_product AS p
INNER JOIN oc_product_description AS pd
    ON p.product_id = pd.product_id
WHERE pd.language_id = 1
    AND p.model >= 1000
GROUP BY
    p.product_id,
    p.model,
    p.quantity,
    p.status,
    pd.name
ORDER BY p.model;
